关于智能循迹小车的一些建议

一、硬件
首先要说的就是硬件的问题了，因为硬件出错大部分情况下比程序出错难改也难排查。我的小车是四轮，两个红外的，有一样的可以参考一下，毕竟我也是新手，踩过不少坑才让我的小车成功跑到26秒。
1、两个红外的间距问题。这个是真的需要根据你的场地仔细调整，因为这会影响它的转弯灵敏度，调好了也可以防抖。如果小车跑起来很抖，可以试着把距离调大一点。调好之后就不要再轻易的改动了，因为你一动，小车可能就又出问题了，包括但不限于转弯不灵敏，抖得很厉害等等，又要花时间去调，别问我怎么知道的。
2.焊接技巧。博主可谓是真的零基础学焊接，焊接的时候一定要先把电烙铁放到要焊接的地方之后再拿锡丝靠近，这样才会容易焊，也焊的更好。
二、程序
1、程序的话，尽量自己写，实在不会了就在网上找一些大神的程序对照自己的硬件改一下参数或让AI帮你写，但是记住要找聪明一点的AI，豆包啥的就不用想了，它写不好的，而且会写的很招笑，别问我怎么知道的。
2、再没啥说的了，贴一段我的代码供大家参考。
#include <reg52.h>
#define IR_BLACK 1
#define PWM_PERIOD 50
#define SPD_RUN_L 46
#define SPD_RUN_R 46
#define SPD_TURN_OUT 50
#define SPD_TURN_IN 38
#define TURN_IN_REV 1
#define TURN_MS 2
#define RAMP_ON 1
#define RAMP_TICK 2

sbit IR_L = P2^1;
sbit IR_R = P2^0;
sbit ENA_L = P1^0;
sbit IN1_L = P1^1;
sbit IN2_L = P1^2;
sbit IN3_R = P1^3;
sbit IN4_R = P1^4;
sbit ENB_R = P1^5;

static unsigned char duty_L = 0, duty_R = 0;
static unsigned char targ_L = 0, targ_R = 0;
static unsigned char pwm_cnt = 0;
static unsigned char ramp_cnt = 0;
static unsigned int sys_ms = 0;
static unsigned char ms_cnt = 0;

static unsigned char turn_dir = 0;
static unsigned int turn_start = 0;
static unsigned char last_sensor = 0;

void Delay_ms(unsigned int ms)
{
unsigned int i;
unsigned char j;
for (i = 0; i < ms; i++)
for (j = 0; j < 200; j++);
}

void Timer0_Init(void)
{
TMOD &= 0xF0;
TMOD |= 0x02;
TH0 = 0x9C;
TL0 = 0x9C;
ET0 = 1;
EA = 1;
TR0 = 1;
}
void Timer0_ISR(void) interrupt 1
{
if (++pwm_cnt >= PWM_PERIOD) pwm_cnt = 0;
ENA_L = (pwm_cnt < duty_L) ? 1 : 0;
ENB_R = (pwm_cnt < duty_R) ? 1 : 0;

if (++ms_cnt >= 10) {
    ms_cnt = 0;
    sys_ms++;
}

#if RAMP_ON
if (++ramp_cnt >= RAMP_TICK) {
ramp_cnt = 0;
if (duty_L < targ_L) duty_L++;
else if (duty_L > targ_L) duty_L–;
if (duty_R < targ_R) duty_R++;
else if (duty_R > targ_R) duty_R–;
}
#endif
}

static void Apply_Speed(void)
{
#if !RAMP_ON
duty_L = targ_L;
duty_R = targ_R;
#endif
}

void Left_Motor(unsigned char dir, unsigned char speed)
{
if (dir) { IN1_L = 1; IN2_L = 0; }
else { IN1_L = 0; IN2_L = 1; }
if (speed > PWM_PERIOD) speed = PWM_PERIOD;
targ_L = speed;
Apply_Speed();
}

void Right_Motor(unsigned char dir, unsigned char speed)
{
if (dir) { IN3_R = 1; IN4_R = 0; }
else { IN3_R = 0; IN4_R = 1; }
if (speed > PWM_PERIOD) speed = PWM_PERIOD;
targ_R = speed;
Apply_Speed();
}

void Car_Stop(void)
{
targ_L = targ_R = 0;
duty_L = duty_R = 0;
IN1_L = 0; IN2_L = 0;
IN3_R = 0; IN4_R = 0;
ENA_L = 0; ENB_R = 0;
}

void Go_Straight(void)
{
Left_Motor(1, SPD_RUN_L);
Right_Motor(1, SPD_RUN_R);
}

void Turn_Left(void)
{
#if TURN_IN_REV
Left_Motor(0, SPD_TURN_IN);
#else
Left_Motor(1, SPD_TURN_IN);
#endif
Right_Motor(1, SPD_TURN_OUT);
}

void Turn_Right(void)
{
Left_Motor(1, SPD_TURN_OUT);
#if TURN_IN_REV
Right_Motor(0, SPD_TURN_IN);
#else
Right_Motor(1, SPD_TURN_IN);
#endif
}

unsigned char Read_IR(void)
{
unsigned char v = 0;
if (IR_L == IR_BLACK) v |= 0x02;
if (IR_R == IR_BLACK) v |= 0x01;
return v;
}

void main(void)
{
unsigned char s1, s2, state;
unsigned int now;
IR_L = 1;
IR_R = 1;
Car_Stop();
Timer0_Init();
Delay_ms(500);
while (1)
{
s1 = Read_IR();
Delay_ms(1);
s2 = Read_IR();
if (s1 == s2) { state = s1; last_sensor = s1; }
else { state = last_sensor; }

    EA = 0; now = sys_ms; EA = 1;
    if (state == 0x03)
    {
        Car_Stop();
        turn_dir = 0;
    }
    else if (state == 0x02)
    {
        turn_dir   = 2;
        turn_start = now;
        Turn_Left();
    }
    else if (state == 0x01)
    {
        turn_dir   = 1;
        turn_start = now;
        Turn_Right();
    }
    else
    {
        if (turn_dir != 0 && (unsigned int)(now - turn_start) < TURN_MS)
        {
            if (turn_dir == 2) Turn_Left();
            else               Turn_Right();
        }
        else
        {
            turn_dir = 0;
            Go_Straight();
        }
    }
    Delay_ms(1);
}

}
